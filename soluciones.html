Código Fuente Python POO (Estructura de Control)

class MotorReductor:
    def __init__(self, posicion: str):
        self.posicion = posicion
        self.velocidad = 0

    def fijar_velocidad(self, pwm: int):
        self.velocidad = pwm
        print(f"Motor {self.posicion}: Velocidad a {self.velocidad} RPM")

class ModuloBluetooth:
    def __init__(self):
        self.conectado = True

    def recibir_comando(self, comando: str) -> str:
        print(f"Comando Bluetooth recibido: {comando}")
        return comando

class BarcoRobot:
    def __init__(self):
        self.motor_izq = MotorReductor("Izquierdo")
        self.motor_der = MotorReductor("Derecho")
        self.bluetooth = ModuloBluetooth()
        self.en_agua = True

    def navegar(self, direccion: str):
        if direccion == "ADELANTE":
            self.motor_izq.fijar_velocidad(50)
            self.motor_der.fijar_velocidad(50)
        elif direccion == "GIRO_IZQUIERDA":
            self.motor_izq.fijar_velocidad(10)
            self.motor_der.fijar_velocidad(50)
        elif direccion == "DETENER":
            self.motor_izq.fijar_velocidad(0)
            self.motor_der.fijar_velocidad(0)
